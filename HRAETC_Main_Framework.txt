    end

x_edge(:,t)-x_est(:,t);

0.85*d_est(:,max(t-1,1))+...
observer_gain*innovation;

    %% -----------------------------------------------------

    % ------------------------------------------------------
    
   phi_est = 0.03*sin(x_est(:,t)) + ...
          0.001*(x_est(:,t).^2);

    % NONLINEAR STATE ESTIMATION
    % ------------------------------------------------------
    
   x_est(:,t+1) = A*x_est(:,t) + ...

               phi_est + ...

B*u+...
0.70*E_hybrid*attack_effect+...
phi_hybrid;

% State saturation

x_hybrid(:,t+1) = ...
max(min(x_hybrid(:,t+1),200),-200);
    
end


%% =========================================================
% SIMULATION : UNCERTAIN RENEWABLE FLUCTUATION
% ==========================================================

for t = 1:T-1
    
    phi_uncertain = ...
    0.03*sin(x_uncertain(:,t)) + ...
    0.001*(x_uncertain(:,t).^2);
    
    
    x_uncertain(:,t+1) = ...
A*x_uncertain(:,t) + ...
B*u + ...
pmu_uncertainty(:,t) + ...
phi_uncertain;

% State saturation

x_uncertain(:,t+1) = ...
max(min(x_uncertain(:,t+1),200),-200);
    
end


%% =========================================================
% SIMULATION : WITH ADAPTIVE CONTROL
% ==========================================================

for t = 1:T-1
    
phi_adaptive = ...
0.03*sin(x_adaptive(:,t)) + ...
0.001*(x_adaptive(:,t).^2);
    
attack_effect = total_attack(1:2,t);

%% Tracking Error

x_ref = 0.01*sin(0.05*t)*ones(4,1);

tracking_error = x_adaptive(:,t) - x_ref;


adaptive_u = ...
-0.36*K_self(:,:,t)*tracking_error + ...
0.42*d_est(1:2,t);

tracking_comp = ...
0.18*(x_adaptive(:,t)-x_normal(:,t));

attack_comp = ...
0.55*E_hybrid*d_est(1:2,t)+...
0.20*tracking_comp;

x_adaptive(:,t+1)=...
0.92*A*x_adaptive(:,t)+...
1.10*B*adaptive_u+...
0.18*E_hybrid*attack_effect-...
attack_comp+...
0.08*phi_adaptive;

x_adaptive(:,t+1)=...
max(min(x_adaptive(:,t+1),200),-200);
   
end
